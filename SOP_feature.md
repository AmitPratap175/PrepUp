"""
Tool wrappers for agent use, including Google Search.

import os
from typing import Optional
from langchain_community.tools import GoogleSearchAPIWrapper
from langchain_core.tools import tool
from src.config import settings


# Initialize Google Search
search_wrapper = GoogleSearchAPIWrapper(
    google_api_key=settings.GOOGLE_API_KEY,
    google_cse_id=settings.GOOGLE_SEARCH_ENGINE_ID,
)


@tool
def google_search(query: str, num_results: int = 3) -> str:
    """
    Search the web using Google Custom Search.
    
    Args:
        query: Search query string
        num_results: Number of results to return (default 3)
    
    Returns:
        String containing search results with titles and snippets
    """
    try:
        results = search_wrapper.results(query, num_results=num_results)
        
        if not results:
            return f"No results found for: {query}"
        
        formatted_results = []
        for result in results:
            formatted_results.append(
                f"Title: {result.get('title', 'N/A')}\n"
                f"Link: {result.get('link', 'N/A')}\n"
                f"Snippet: {result.get('snippet', 'N/A')}\n"
            )
        
        return "\n---\n".join(formatted_results)
    
    except Exception as e:
        return f"Error searching: {str(e)}"


# Export for use in agents
TOOLS = [google_search]
TOOLS_MAP = {tool.name: tool for tool in TOOLS}

"""
Base utilities for all agents.
Handles model initialization, tool binding, and common patterns.
"""

import json
import re
from typing import Any
from langchain_google_genai import ChatGoogleGenerativeAI
from langchain_core.messages import BaseMessage, HumanMessage
from src.models import GraphState, SOPForgeConfig
from src.config import settings
from src.utils.tools import TOOLS


def get_model(model_name: str, temperature: float = 0.7) -> ChatGoogleGenerativeAI:
    """
    Initialize a Google Gemini model with tool binding if needed.
    
    Args:
        model_name: Model identifier (e.g., 'gemini-2.5-flash')
        temperature: Sampling temperature (0.0-1.0)
    
    Returns:
        Initialized ChatGoogleGenerativeAI model
    """
    return ChatGoogleGenerativeAI(
        model=model_name,
        api_key=settings.GEMINI_API_KEY,
        temperature=temperature,
        max_tokens=2000,
    )


def get_model_with_tools(model_name: str, temperature: float = 0.7) -> ChatGoogleGenerativeAI:
    """
    Initialize a model with tools bound for agent use.
    
    Args:
        model_name: Model identifier
        temperature: Sampling temperature
    
    Returns:
        Model with tools bound
    """
    model = get_model(model_name, temperature)
    return model.bind_tools(TOOLS)


def extract_json_from_response(response_text: str) -> dict:
    """
    Extract JSON object from LLM response.
    Handles cases where model wraps JSON in markdown or text.
    
    Args:
        response_text: Raw model response
    
    Returns:
        Parsed JSON object
    
    Raises:
        ValueError: If no valid JSON found
    """
    # Try direct parse first
    try:
        return json.loads(response_text)
    except json.JSONDecodeError:
        pass
    
    # Try extracting from markdown code blocks
    json_match = re.search(r"```(?:json)?\s*(\{.*?\})\s*```", response_text, re.DOTALL)
    if json_match:
        try:
            return json.loads(json_match.group(1))
        except json.JSONDecodeError:
            pass
    
    # Try finding raw JSON object
    json_match = re.search(r"\{.*\}", response_text, re.DOTALL)
    if json_match:
        try:
            return json.loads(json_match.group(0))
        except json.JSONDecodeError:
            pass
    
    raise ValueError(f"Could not extract valid JSON from response: {response_text[:200]}")


def format_state_for_context(state: GraphState) -> str:
    """
    Format relevant parts of state for LLM context in prompts.
    
    Args:
        state: Current graph state
    
    Returns:
        Formatted context string
    """
    context = f"""
## User Information
**Goal:** {state['user_prompt']}

**Profile:**
{state['user_profile']}
"""
    
    if state['research_findings']:
        context += f"""
## Research Findings
{state['research_findings']}
"""
    
    if state['draft']:
        context += f"""
## Current Draft
{state['draft']}
"""
    
    if state['review_notes']:
        context += f"""
## Review Feedback
{state['review_notes']}
"""
    
    return context


def safe_invoke_model(
    model: ChatGoogleGenerativeAI,
    prompt: str,
    system_message: str = "",
) -> str:
    """
    Safely invoke model with error handling.
    
    Args:
        model: LLM to invoke
        prompt: User/input prompt
        system_message: Optional system message/instructions
    
    Returns:
        Model response text
    
    Raises:
        RuntimeError: If model invocation fails
    """
    try:
        messages = []
        if system_message:
            messages.append({"role": "system", "content": system_message})
        messages.append({"role": "user", "content": prompt})
        
        response = model.invoke(messages)
        return response.content if hasattr(response, 'content') else str(response)
    
    except Exception as e:
        raise RuntimeError(f"Model invocation failed: {str(e)}")

"""

    """
    Planner node: Generate research queries based on user goal and profile.

        - user_prompt: User's SOP goal
        - user_profile: User's background
    
    Outputs:
        - research_plan: List of search queries

"""
ResearcherAgent: Executes web searches and synthesizes findings.
Role: "Context Gatherer" - finds specific, credible information.
"""

from typing import Optional
from langchain_community.tools import GoogleSearchAPIWrapper
from src.models import GraphState, SOPForgeConfig
from src.agents.base import (
    get_model,
    format_state_for_context,
    safe_invoke_model,
)
from src.utils.prompts import RESEARCHER_SYSTEM_PROMPT
from src.config import settings


def run_researcher(state: GraphState) -> dict:
    """
    Researcher node: Execute web searches and synthesize findings.
    
    Inputs:
        - research_plan: List of search queries
    
    Outputs:
        - research_findings: Synthesized research summary
    
    Args:
        state: Current graph state
    
    Returns:
        Updated state dict with research_findings
    """
    
    research_plan = state.get("research_plan", [])
    
    if not research_plan:
        return {
            "research_findings": "No research queries were generated.",
            "messages": [{"role": "assistant", "content": "No research queries to execute."}]
        }
    
    # Execute searches
    search_results = _execute_searches(research_plan)
    
    if not search_results.strip():
        return {
            "research_findings": "No search results found.",
            "messages": [{"role": "assistant", "content": "Web search returned no results."}]
        }
    
    # Synthesize findings with LLM
    model = get_model(SOPForgeConfig.REVIEWER_MODEL, temperature=0.5)
    
    prompt = f"""You are synthesizing web search results into a coherent research summary.

## Search Results
{search_results}

## Task
Compile these results into a clear, organized research summary (500-800 words) that highlights:
1. Key faculty and their research interests
2. Program structure and values
3. Recent publications or achievements
4. Unique opportunities or resources
5. Program-specific strengths

Write the summary now, making it suitable for personalizing an SOP.
"""
    
    research_findings = safe_invoke_model(
        model,
        prompt,
        system_message=RESEARCHER_SYSTEM_PROMPT,
    )
    
    return {
        "research_findings": research_findings,
        "messages": [{"role": "assistant", "content": "Researcher synthesized web findings."}]
    }


def _execute_searches(queries: list[str]) -> str:
    """
    Execute web searches for all queries and format results.
    
    Args:
        queries: List of search queries
    
    Returns:
        Formatted search results string
    """
    search_wrapper = GoogleSearchAPIWrapper(
        google_api_key=settings.GOOGLE_API_KEY,
        google_cse_id=settings.GOOGLE_SEARCH_ENGINE_ID,
    )
    
    all_results = []
    
    for query in queries:
        try:
            results = search_wrapper.results(
                query,
                num_results=SOPForgeConfig.SEARCH_RESULTS_PER_QUERY
            )
            
            query_results = f"\n### Query: {query}\n"
            for i, result in enumerate(results, 1):
                query_results += (
                    f"\n**Result {i}:**\n"
                    f"Title: {result.get('title', 'N/A')}\n"
                    f"Link: {result.get('link', 'N/A')}\n"
                    f"Snippet: {result.get('snippet', 'N/A')}\n"
                )
            
            all_results.append(query_results)
        
        except Exception as e:
            print(f"Warning: Search failed for '{query}': {e}")
            continue
    
    return "\n".join(all_results) if all_results else ""

"""
Conditional edge routing functions for the LangGraph.
These functions determine which node to visit next based on state.
"""

from typing import Literal
from src.models import GraphState, SOPForgeConfig


def check_for_revisions(state: GraphState) -> Literal["editor", "human_in_the_loop"]:
    """
    Route based on reviewer's feedback.
    
    If review_notes == "PERFECT", skip to human checkpoint.
    Otherwise, go to editor for revisions.
    
    Args:
        state: Current graph state
    
    Returns:
        Node name to route to
    """
    review_notes = state.get("review_notes", "").strip().upper()
    
    if review_notes == "PERFECT":
        return "human_in_the_loop"
    else:
        return "editor"


def check_iteration_count(state: GraphState) -> Literal["reviewer", "human_in_the_loop"]:
    """
    Route based on iteration count to prevent infinite loops.
    
    If iteration_count > MAX_ITERATIONS, skip to human.
    Otherwise, loop back to reviewer.
    
    Args:
        state: Current graph state
    
    Returns:
        Node name to route to
    """
    iteration_count = state.get("iteration_count", 0)
    
    if iteration_count >= SOPForgeConfig.MAX_ITERATIONS:
        return "human_in_the_loop"
    else:
        return "reviewer"


def check_human_feedback(state: GraphState) -> Literal["editor", "__end__"]:
    """
    Route based on human feedback at HITL checkpoint.
    
    If human_feedback == "APPROVE", end.
    Otherwise, route back to editor with new instructions.
    
    Args:
        state: Current graph state
    
    Returns:
        Node name to route to
    """
    human_feedback = state.get("human_feedback", "").strip().upper()
    
    if human_feedback == "APPROVE":
        return "__end__"
    else:
        return "editor"

```python
"""
LangGraph builder: Constructs the complete SOP generation graph.
Assembles all nodes, edges, and creates the compiled graph.
"""

from langgraph.graph import StateGraph, START, END
from src.models import GraphState
from src.graph.nodes import (
    node_planner,
    node_researcher,
    node_writer,
    node_reviewer,
    node_editor,
    node_human_in_the_loop,
)
from src.graph.edges import (
    check_for_revisions,
    check_iteration_count,
    check_human_feedback,
)
from src.graph.checkpointer import get_checkpointer


def build_sop_graph():
    """
    Build and compile the SOPForge LangGraph.
    
    Architecture:
    START
      ↓
    planner → researcher → writer → reviewer ↓
                                    ├─→ [PERFECT?]
                                    │    ↓
                                    └→ editor → [MAX_ITERATIONS?]
                                                ├→ reviewer (loop)
                                                └→ human_in_the_loop
                                    ↓
                              human_in_the_loop
                                    ↓
                              [APPROVE?]
                                    ├→ END
                                    └→ editor (revise)
                                         ↓
                                       reviewer
    
    Returns:
        Compiled LangGraph ready for invocation
    """
    
    # Initialize graph with state schema
    graph = StateGraph(GraphState)
    
    # Add all nodes
    graph.add_node("planner", node_planner)
    graph.add_node("researcher", node_researcher)
    graph.add_node("writer", node_writer)
    graph.add_node("reviewer", node_reviewer)
    graph.add_node("editor", node_editor)
    graph.add_node("human_in_the_loop", node_human_in_the_loop)
    
    # Add edges (linear flow until reviewer)
    graph.add_edge(START, "planner")
    graph.add_edge("planner", "researcher")
    graph.add_edge("researcher", "writer")
    graph.add_edge("writer", "reviewer")
    
    # Conditional edge after reviewer
    graph.add_conditional_edges(
        "reviewer",
        check_for_revisions,
        {
            "editor": "editor",
            "human_in_the_loop": "human_in_the_loop",
        }
    )
    
    # Conditional edge after editor (iteration check)
    graph.add_conditional_edges(
        "editor",
        check_iteration_count,
        {
            "reviewer": "reviewer",
            "human_in_the_loop": "human_in_the_loop",
        }
    )
    
    # Conditional edge after human-in-the-loop
    graph.add_conditional_edges(
        "human_in_the_loop",
        check_human_feedback,
        {
            "editor": "editor",
            "__end__": END,
        }
    )
    
    # Compile with checkpointer (required for interrupts)
    checkpointer = get_checkpointer()
    compiled_graph = graph.compile(
        checkpointer=checkpointer,
        interrupt_before=["human_in_the_loop"],  # Pause before HITL node
    )
    
    return compiled_graph

```python
"""
SOPForge main entry point and orchestration logic.
Handles the workflow with human-in-the-loop support.
"""

import uuid
from src.models import GraphState, create_initial_state
from src.graph.builder import build_sop_graph
from src.config import settings
from langgraph.types import Command


class SOPForgeOrchestrator:
    """Main orchestrator for SOPForge workflow."""
    
    def __init__(self):
        """Initialize the orchestrator and build graph."""
        settings.validate()  # Validate required env vars
        self.graph = build_sop_graph()
    
    def generate_sop(
        self,
        user_prompt: str,
        user_profile: str,
        thread_id: str = None,
    ) -> dict:
        """
        Generate an SOP from user inputs with human-in-the-loop approval.
        
        Runs the full SOP generation pipeline and pauses at human review checkpoint.
        Call submit_feedback() after to resume with user's decision.
        
        Args:
            user_prompt: User's SOP goal (e.g., "MS in CS at Stanford, focus on AI")
            user_profile: User's background and resume
            thread_id: Optional thread ID for persistence (generated if not provided)
        
        Returns:
            Dict with status, draft, and thread_id
        """
        
        if not thread_id:
            thread_id = str(uuid.uuid4())
        
        # Create initial state
        initial_state = create_initial_state(user_prompt, user_profile)
        config = {"configurable": {"thread_id": thread_id}}
        
        print(f"\n🚀 Starting SOP generation (thread: {thread_id})...")
        print("📋 Running: Planner → Researcher → Writer → Reviewer...")
        
        try:
            # Run graph until interrupted (human_in_the_loop)
            result = self.graph.invoke(initial_state, config=config)
            
            return {
                "status": "paused_at_human_review",
                "thread_id": thread_id,
                "draft": result.get("draft", ""),
                "iteration_count": result.get("iteration_count", 0),
            }
        
        except Exception as e:
            return {
                "status": "error",
                "thread_id": thread_id,
                "error": str(e),
            }
    
    def submit_feedback(
        self,
        thread_id: str,
        feedback: str,
    ) -> dict:
        """
        Submit user feedback at human-in-the-loop checkpoint.
        
        Resumes graph execution with user's decision:
        - "APPROVE": Accept draft and end
        - Anything else: Treat as revision instructions and loop back to editor
        
        Args:
            thread_id: Thread ID from generate_sop()
            feedback: User feedback ("APPROVE" or revision instructions)
        
        Returns:
            Dict with status and final draft (if approved)
        """
        
        config = {"configurable": {"thread_id": thread_id}}
        
        try:
            # Resume with user feedback
            result = self.graph.invoke(
                Command(resume=feedback),
                config=config,
            )
            
            return {
                "status": "completed" if feedback.upper() == "APPROVE" else "revising",
                "thread_id": thread_id,
                "draft": result.get("draft", ""),
                "iteration_count": result.get("iteration_count", 0),
            }
        
        except Exception as e:
            return {
                "status": "error",
                "thread_id": thread_id,
                "error": str(e),
            }
    
    def stream_generation(
        self,
        user_prompt: str,
        user_profile: str,
        thread_id: str = None,
    ):
        """
        Stream the SOP generation process (updates only).
        
        Yields state updates as each node executes.
        Note: Still pauses at human_in_the_loop for approval.
        
        Args:
            user_prompt: User's SOP goal
            user_profile: User's background
            thread_id: Optional thread ID
        
        Yields:
            State updates as dict
        """
        
        if not thread_id:
            thread_id = str(uuid.uuid4())
        
        initial_state = create_initial_state(user_prompt, user_profile)
        config = {"configurable": {"thread_id": thread_id}}
        
        try:
            for chunk in self.graph.stream(
                initial_state,
                config=config,
                stream_mode="updates",
            ):
                yield chunk
        
        except Exception as e:
            yield {"error": str(e)}
    
    def get_draft(self, thread_id: str) -> str:
        """Retrieve current draft for a thread."""
        config = {"configurable": {"thread_id": thread_id}}
        try:
            state = self.graph.get_state(config)
            return state.values.get("draft", "")
        except Exception as e:
            return f"Error retrieving draft: {str(e)}"

    if _orchestrator is None:
        _orchestrator = SOPForgeOrchestrator()
    return _orchestrator

# SOPForge 🚀

**Multi-Agent System for Generating Compelling Statements of Purpose**

SOPForge is an intelligent, multi-agent system built with LangChain and LangGraph that helps users craft personalized, impactful Statements of Purpose (SOPs) for academic and professional applications.

## Overview

SOPForge coordinates a team of specialized AI agents:

- **Planner**: Analyzes user goals and generates targeted research queries
- **Researcher**: Executes web searches and synthesizes findings
- **Writer**: Crafts compelling, personalized SOP narratives
- **Reviewer**: Provides constructive feedback on draft quality
- **Editor**: Implements revisions based on feedback
- **Human-in-the-Loop**: Pauses for user approval/feedback before completion

## Features

✨ **Multi-Agent Collaboration** - Specialized agents work together for better results

## Architecture

### Graph Flow

```
START
  ↓
Planner (generates research queries)
  ↓
Researcher (executes web searches, synthesizes findings)
  ↓
Writer (crafts compelling narrative)
  ↓
Reviewer (evaluates quality)
  ├─→ PERFECT? → Human Review Checkpoint
  │
  └─→ Needs Revision?
       ↓
       Editor (implements feedback)
       ↓
       [Check iteration count]
       ├─→ Max reached? → Human Review Checkpoint
       └─→ Loop back to Reviewer
       
Human Review Checkpoint
  ├─→ "APPROVE" → END
  └─→ Feedback provided → Editor (revise) → Reviewer
```

### State Management

All agents share a `GraphState` containing:
- User inputs (prompt, profile)
- Generated content (research plan, findings, draft)
- Feedback (review notes, human feedback)
- Control flow (iteration count)

## Troubleshooting

### API Keys Not Working

### Graph Execution Hangs

1. Check LangSmith logs if enabled
2. Verify all API keys have sufficient quota
3. Try with a simpler `user_prompt` for testing

### Memory Issues with Large Drafts

1. Increase `MAX_DRAFT_TOKENS` in `src/models.py`
2. Switch to PostgreSQL checkpointer for production
3. Run on machine with more available RAM

### **2️⃣9️⃣ `run.py`** - Quick Start Script

```python
"""
Quick start script for SOPForge.
Run this to test the system locally.
"""

import sys
from src.main import get_orchestrator
from src.config import settings


def main():
    """Run SOPForge in interactive mode."""
    
    print("=" * 70)
    print("SOPForge - Multi-Agent SOP Generation System")
    print("=" * 70)
    
    # Get user inputs
    print("\n📝 Please provide your information:\n")
    
    user_prompt = input("Your SOP goal (e.g., 'MS in CS at Stanford, focus on AI'): ").strip()
    if not user_prompt:
        print("Error: SOP goal cannot be empty")
        sys.exit(1)
    
    print("\nEnter your profile/background (press Enter twice when done):")
    lines = []
    while True:
        line = input()
        if not line:
            if lines:
                break
        else:
            lines.append(line)
    
    user_profile = "\n".join(lines)
    if not user_profile:
        print("Error: Profile cannot be empty")
        sys.exit(1)
    
    # Initialize orchestrator
    print("\n🚀 Starting SOP generation...\n")
    try:
        settings.validate()
    except ValueError as e:
        print(f"❌ Configuration error: {e}")
        sys.exit(1)
    
    orchestrator = get_orchestrator()
    
    # Generate SOP
    result = orchestrator.generate_sop(user_prompt, user_profile)
    
    if result["status"] == "error":
        print(f"❌ Error: {result['error']}")
        sys.exit(1)
    
    # Display draft
    print("\n✅ Initial draft completed!")
    print("=" * 70)
    print("\n📄 DRAFT:\n")
    print(result["draft"])
    print("\n" + "=" * 70)
    
    # Human review loop
    thread_id = result["thread_id"]
    iteration = 0
    
    while True:
        print(f"\n🔄 Review Iteration {iteration + 1}")
        feedback = input(
            "Your feedback (type 'APPROVE' to accept, or provide revision feedback): "
        ).strip()
        
        if not feedback:
            print("Please provide feedback or type 'APPROVE'")
            continue
        
        result = orchestrator.submit_feedback(thread_id, feedback)
        
        if result["status"] == "error":
            print(f"❌ Error: {result['error']}")
            break
        
        print("\n📄 UPDATED DRAFT:\n")
        print(result["draft"])
        
        if result["status"] == "completed":
            print("\n" + "=" * 70)
            print("✨ SOP generation complete!")
            print("=" * 70)
            break
        
        iteration += 1
        if iteration >= 5:
            print("\n⚠️  Reached maximum revision iterations")
            break


if __name__ == "__main__":
    main()
```

def create_sopforge_package():
    """Create a complete zipped SOPForge package."""
    
    print("🚀 Creating SOPForge Package...")
    print("=" * 70)
    
    # Define what to include/exclude
    INCLUDE_DIRS = [
        "src",
        "tests",
    ]
    
    INCLUDE_FILES = [
        ".env.example",
        ".gitignore",
        "Dockerfile",
        "docker-compose.yml",
        "Makefile",
        "README.md",
        "SETUP.md",
        "langgraph.json",
        "pyproject.toml",
        "requirements.txt",
        "run.py",
        "package.py",
    ]
    
    # Create zip filename with timestamp
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    zip_filename = f"sopforge_{timestamp}.zip"
    
    # Create temporary directory for staging
    staging_dir = Path("sopforge_staging")
    if staging_dir.exists():
        shutil.rmtree(staging_dir)
    staging_dir.mkdir()
    
    try:
        print(f"\n📁 Staging files to {staging_dir}...\n")
        
        # Copy directories
        for dir_name in INCLUDE_DIRS:
            src_path = Path(dir_name)
            if src_path.exists():
                dest_path = staging_dir / dir_name
                shutil.copytree(src_path, dest_path, ignore=shutil.ignore_patterns("__pycache__", "*.pyc", ".pytest_cache"))

        # Copy files
        for file_name in INCLUDE_FILES:
            src_path = Path(file_name)
            if src_path.exists():
                dest_path = staging_dir / file_name
                shutil.copy2(src_path, dest_path)
                print(f"  ✓ {file_name}")

SOPForge is a multi-agent LLM system orchestrated by LangGraph. It follows a pipeline architecture where each agent specializes in a specific task.

```
┌─────────────────────────────────────────────────────────────────┐
│                      SOPForge Pipeline                          │
├─────────────────────────────────────────────────────────────────┤
│                                                                  │
│  START                                                           │
│    ↓                                                             │
│  ┌──────────────┐        ┌──────────────┐        ┌────────────┐│
│  │   Planner    │ ──────→│ Researcher   │ ──────→│   Writer   ││
│  │  Generate    │        │  Web Search  │        │   Draft    ││
│  │  Queries     │        │  Synthesize  │        │   Craft    ││
│  └──────────────┘        └──────────────┘        └────────────┘│
│                                                        ↓        │
│                                              ┌──────────────┐   │
│                                              │   Reviewer   │   │
│                                              │   Evaluate   │   │
│                                              │   Quality    │   │
│                                              └──────────────┘   │
│                                                ↓         ↓      │
│                                         PERFECT?    REVISE?    │
│                                            ↓            ↓      │
│  ┌──────────────────────────────────┐     │      ┌──────────────┐
│  │   Human-in-the-Loop (HITL)       │←────┘      │   Editor     │
│  │   - Display Draft                │            │   Revise     │
│  │   - Wait for User Feedback       │            │   Refactor   │
│  │   - APPROVE or NEW FEEDBACK      │            └──────────────┘
│  └──────────────────────────────────┘                  ↑
│           ↓              ↓                             │
│        APPROVE      FEEDBACK───────────────────────────┘
│           ↓
│          END
│
└─────────────────────────────────────────────────────────────────┘
```

## Component Details

### 1. Planner Agent (`src/agents/planner.py`)

**Role**: Research Strategist

**Input**: 
- `user_prompt`: SOP goal (e.g., "MS in CS at Stanford, AI focus")
- `user_profile`: Background and achievements

**Process**:
1. Analyzes user's goal and background
2. Identifies key information needs
3. Generates 3-5 targeted search queries

**Output**: `research_plan` - List of search queries

**Model**: Gemini 2.5 Flash (fast, cost-effective)

**Example Output**:
```
[
  "Stanford CS AI faculty research interests 2024",
  "Stanford machine learning lab recent publications",
  "MS Computer Science Stanford curriculum AI specialization",
  "Stanford AI index lab Dario Amodei",
]
```

### 2. Researcher Agent (`src/agents/researcher.py`)

**Role**: Information Gatherer

**Input**: `research_plan` from Planner

**Process**:
1. Executes Google Custom Search for each query
2. Collects results (3 per query)
3. Uses Gemini to synthesize into coherent summary
4. Extracts key facts about program, faculty, labs

**Output**: `research_findings` - Synthesized research (500-800 words)

**External Service**: Google Custom Search API

**Example Output**:
```
Stanford CS Program Overview:
The MS in Computer Science at Stanford emphasizes machine learning 
and artificial intelligence. Key faculty include [...]

Faculty & Research:
Prof. Fei-Fei Li leads the HAI lab focusing on human-centered AI [...]

Recent Publications:
The lab published 12 papers in 2024 on vision transformers [...]
```

### 3. Writer Agent (`src/agents/writer.py`)

**Role**: Creative Narrative Writer

**Input**:
- `user_prompt`: Goal
- `user_profile`: Background
- `research_findings`: Program details

**Process**:
1. Analyzes all inputs
2. Crafts compelling opening hook
3. Weaves background with program fit
4. Creates coherent narrative arc
5. Includes specific details from research

**Output**: `draft` - Complete SOP (1000-1500 words)

**Model**: Gemini 1.5 Pro (higher quality, more reasoning)

**Example Structure**:
```
1. Opening Hook (100 words)
   - Compelling motivation

2. Background & Achievements (300 words)
   - Education, experience, projects
   - Key skills and accomplishments

3. Why This Program (400 words)
   - Specific faculty alignment
   - Lab interests match
   - Curriculum fit

4. Future Vision (200 words)
   - Career goals
   - Research interests
   - Impact aspirations

5. Closing (100 words)
   - Confident conclusion
   - Enthusiasm
```

### 4. Reviewer Agent (`src/agents/reviewer.py`)

**Role**: Quality Assurance Critic

**Input**: 
- `draft`: Current SOP
- `user_prompt`: Goal
- `user_profile`: Background
- `research_findings`: Context

**Process**:
1. Evaluates against rubric:
   - Goal alignment
   - Profile integration
   - Research specificity
   - Narrative quality
   - Tone & grammar
   - Uniqueness

2. Returns either:
   - "PERFECT" (excellent)
   - Bulleted list of specific improvements

**Output**: `review_notes` - Feedback string

**Model**: Gemini 2.5 Flash (temperature=0.3 for consistency)

**Example Output - Perfect**:
```
PERFECT
```

**Example Output - Needs Revision**:
```
- The opening is generic; start with a specific moment or insight
- Missing explicit mention of Prof. Pieter Abbeel's robotics work
- The middle section could better connect your Google experience to Stanford's goals
- Final paragraph feels rushed; expand on long-term vision
```

### 5. Editor Agent (`src/agents/editor.py`)

**Role**: Revisions Specialist

**Input**:
- `draft`: Current SOP
- `review_notes` OR `human_feedback`: What to improve
- `iteration_count`: How many revisions done

**Process**:
1. Prioritizes human feedback over auto-review
2. Makes targeted improvements (not full rewrite)
3. Preserves working parts of draft
4. Adds specificity where lacking
5. Improves flow and transitions

**Output**: Updated `draft`

**Model**: Gemini 1.5 Pro (temperature=0.6 for creativity)

### 6. Human-in-the-Loop (HITL) (`src/graph/nodes.py`)

**Role**: Final Decision Maker

**Function**: `node_human_in_the_loop`

**Process**:
1. Uses LangGraph's `interrupt()` to pause execution
2. Displays current draft to user
3. Waits indefinitely for user response
4. User chooses:
   - "APPROVE" → End process
   - Feedback text → Route to Editor for revisions

**State Persistence**: Uses checkpointer (MemorySaver or PostgreSQL)

**Thread ID**: Unique identifier for resumable conversations

## State Management

### GraphState Structure

```python
class GraphState(TypedDict):
    # Inputs
    user_prompt: str                    # "MS in CS at Stanford, focus on AI"
    user_profile: str                   # Resume/background
    
    # Generated Content
    research_plan: list[str]            # Search queries
    research_findings: str              # Synthesized research
    draft: str                          # Current SOP
    review_notes: str                   # Reviewer feedback
    
    # Control Flow
    iteration_count: int                # Revision count
    human_feedback: Optional[str]       # User input at HITL
    
    # Observability
    messages: Annotated[list, add_messages]  # LLM traces
```

### State Flow Example

```
Initial:
{
  user_prompt: "MS in CS at Stanford",
  user_profile: "BS CS, 3 years at Google",
  research_plan: [],
  research_findings: "",
  draft: "",
  review_notes: "",
  iteration_count: 0,
  human_feedback: None,
  messages: []
}

After Planner:
{
  ...,
  research_plan: ["Stanford CS faculty AI", "Stanford ML lab", ...],
  ...
}

After Researcher:
{
  ...,
  research_findings: "Stanford CS program emphasizes AI. Key faculty: ...",
  ...
}

After Writer:
{
  ...,
  draft: "I have always been fascinated by artificial intelligence...",
  review_notes: "",
  ...
}

After Reviewer (needs revision):
{
  ...,
  review_notes: "- Opening is too generic\n- Missing specific faculty mentions",
  ...
}

After Editor:
{
  ...,
  draft: "[Updated version addressing feedback]",
  iteration_count: 1,
  ...
}

After Human Review (approved):
{
  ...,
  human_feedback: "APPROVE",
  ... → END
}
```

## Conditional Routing

### check_for_revisions()

After Reviewer:
- If `review_notes == "PERFECT"` → Go to HITL
- Else → Go to Editor

### check_iteration_count()

After Editor:
- If `iteration_count >= MAX_ITERATIONS` → Go to HITL (force user decision)
- Else → Loop back to Reviewer

### check_human_feedback()

After HITL:
- If `human_feedback == "APPROVE"` → END
- Else → Go to Editor with feedback

## Model Selection

| Agent | Model | Temperature | Reason |
|-------|-------|-------------|--------|
| Planner | Gemini 2.5 Flash | 0.5 | Fast, cost-effective, structured output |
| Researcher | Gemini 2.5 Flash | 0.5 | Quick synthesis of results |
| Writer | Gemini 1.5 Pro | 0.7 | Higher quality, creative writing |
| Reviewer | Gemini 2.5 Flash | 0.3 | Consistent, deterministic evaluation |
| Editor | Gemini 1.5 Pro | 0.6 | Balanced creativity and consistency |

## Persistence & Checkpointing

### How It Works

1. **Memory Checkpointer** (Development)
   - Stores state in RAM
   - Fast, perfect for local development
   - Lost on process restart

2. **PostgreSQL Checkpointer** (Production)
   - Stores state in database
   - Survives process restarts
   - Required for production deployment

### Thread ID

Each conversation gets a unique thread ID:
```python
config = {"configurable": {"thread_id": "uuid-1234"}}
```

Same thread ID can:
- Resume interrupted graphs
- Access conversation history
- Maintain state across requests

## Error Handling

### Graceful Degradation

1. **API Key Missing** → `settings.validate()` fails at startup
2. **Search Returns Nothing** → Researcher uses fallback generic responses
3. **Model Rejection** → Returns empty draft with error message
4. **JSON Parse Failure** → Falls back to default/generic output

### Retry Strategy

- Network errors: Automatic retry with exponential backoff
- Rate limits: Handled by SDK
- Invalid responses: Logged and fallback used

## Scalability

### Limitations

- **Sequential Processing**: Agents run one-after-another (not parallel)
- **Web Search Rate Limits**: Google limits requests per second
- **LLM Rate Limits**: API quotas vary by plan
- **Memory**: State kept in memory during execution

### Scaling Solutions

1. **Parallel Researcher Queries** (Future)
   - Execute searches concurrently
   - Reduce researcher time from 30s to 10s

2. **Caching** (Future)
   - Cache research findings by topic
   - Reuse across similar SOPs

3. **Distributed Processing** (LangSmith Deployment)
   - Run multiple instances behind load balancer
   - Handle concurrent users

4. **Database Persistence**
   - Switch to PostgreSQL for larger deployments
   - Track usage metrics and costs

## Security Considerations

### API Keys

- **Storage**: Environment variables via `.env`
- **Never commit**: Listed in `.gitignore`
- **Access**: Only loaded in `src/config.py`
- **Exposure**: Never logged or returned to user

### Input Validation

- User inputs sanitized before LLM calls
- File operations restricted to temp directories
- No shell command execution

### Output Safety

- Generated SOPs don't contain sensitive data
- No PII leaked between users (separate threads)
- State isolated per thread ID

## Key Concepts

### State-Driven Architecture
- All agents operate on shared `GraphState`
- Changes propagate through the pipeline
- Each agent can read full history
- Enables complex workflows

### Tool-Calling LLMs
- Agents have access to tools (Google Search)
- LLMs decide when to use tools
- Automatic execution of tool calls
- Results fed back to LLM

### Human-in-the-Loop
- Graph pauses at checkpoint
- Waits indefinitely for user
- State persisted to database
- Resume with new input

### Iterative Refinement
- Reviewer → Editor feedback loop
- Max iterations prevent infinite loops
- User can override with approval
- Better quality through multiple passes

## Performance Optimization Tips

### Reduce Generation Time
1. Use simpler `user_prompt` for testing
2. Reduce `RESEARCH_QUERIES_LIMIT` (fewer searches)
3. Lower `MAX_DRAFT_TOKENS` (shorter outputs)

### Reduce Costs
1. Use cheaper models (Gemini 2.5 Flash for all agents)
2. Reduce token limits
3. Cache research findings across users
4. Batch process multiple SOPs

### Improve Quality
1. Increase `MAX_ITERATIONS` (more reviews)
2. Use Pro models (Gemini 1.5 Pro)
3. Customize prompts in `src/utils/prompts.py`
4. Add domain-specific research queries

## FAQ

**Q: Can I use different LLM providers?**
A: Yes! Edit `src/agents/base.py` to use OpenAI, Anthropic, etc. instead of Gemini.

**Q: How do I limit costs?**
A: Use cheaper models (Gemini 2.5 Flash), reduce token limits, cache results.

**Q: Can I deploy to AWS/GCP/Azure?**
A: Yes! Use Docker (`docker build -t sopforge .`) then deploy to any container platform.

**Q: How do I add my own research sources?**
A: Add tools to `src/utils/tools.py` and update Researcher to use them.

**Q: Can multiple users use this simultaneously?**
A: Yes, with PostgreSQL checkpointer and multiple API server instances.

**Q: How do I make the output more/less verbose?**
A: Edit token limits in `src/models.py` or adjust prompts in `src/utils/prompts.py`.

**Q: Can I use this for other documents (not just SOPs)?**
A: Yes! It's a general multi-agent generation framework. Customize prompts for resumes, essays, etc.
